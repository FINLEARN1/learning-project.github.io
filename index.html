{ "name": "finlearn1-data-server", "version": "1.0.0", "description": "Serve FINANCIAL_DATA JSON for FinLearn app", "main": "server.js", "scripts": { "start": "node server.js", "start:dev": "NODE_ENV=development node server.js" }, "author": "FinLearn", "license": "MIT", "dependencies": { "express": "^4.18.2" } } // server.js // Start -> reads financial_data.json and exposes it at /financial_data.json // Usage: npm install && npm start const fs = require('fs'); const path = require('path'); const express = require('express'); const PORT = process.env.PORT || 3000; const DATA_FILE = path.join(__dirname, 'financial_data.json'); const app = express(); app.get('/', (req, res) => { res.send('FinLearn data server is running. Endpoint: /financial_data.json'); }); app.get('/financial_data.json', (req, res) => { fs.readFile(DATA_FILE, 'utf8', (err, data) => { if (err) { console.error('Error reading JSON file:', err); return res.status(500).json({ error: 'Unable to load data file' }); } try { const json = JSON.parse(data); res.json(json); } catch (parseErr) { console.error('JSON parse error:', parseErr); res.status(500).json({ error: 'Invalid JSON file' }); } }); }); app.listen(PORT, () => { console.log('--- START ---'); console.log(`FinLearn data server listening on http://localhost:${PORT}`); console.log(`Serving file: ${DATA_FILE}`); console.log('--- END ---'); });
